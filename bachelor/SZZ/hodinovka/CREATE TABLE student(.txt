CREATE TABLE student(
	id BIGINT PRIMARY KEY,
	jmeno VARCHAR(255),
	datum_narozeni DATE,
	program_id BIGINT FOREIGN KEY(references program)
)

CREATE TABLE predmet(
	id BIGINT PRIMARY KEY,
	nazev VARCHAR(255),
	max_pocet_studentu INT,
	rocnik INT,
	semestr VARCHAR(2)

CREATE TABLE student_predmet(
	id BIGINT PRIMARY,
	student_id BIGINT FOREIGN,
	predmet_id BIGINT FOREIGN
)

CREATE TABLE program(
	id BIGINT PRIMARY,
	nazev VARCHAR(255),
	doba_trvani INT
)

//všichni studenti a jména předmětů, které studují
SELECT s.jmeno, p.nazev
FROM student s 
LEFT JOIN student_predmet sp ON sp.student_id = s.id
LEFT JOIN predmet p ON sp.predmet_id = p.id


//studijní programy a počty studentů v nich
SELECT pro.nazev, COUNT(s.id)
FROM program pro
LEFT JOIN student s ON s.program_id = pro.id
GROUP BY s.program_id

//seskupení předmětů podle ročníku studia
SELECT p.rocnik, STRING_AGG(p.nazev, ',')
FROM predmet p
GROUP BY p.rocnik
